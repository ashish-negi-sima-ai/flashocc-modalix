cmake_minimum_required(VERSION 3.16)
project(flashocc_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SimaNeat REQUIRED CONFIG)
add_subdirectory(../cpp pooling)
add_executable(flashocc_replay main.cpp)
target_link_libraries(flashocc_replay PRIVATE flashocc_pool SimaNeat::sima_neat)
target_compile_options(flashocc_replay PRIVATE -Wall -Wextra -Wpedantic)
# The SDK OpenCV package config references optional modules absent from its sysroot.
find_path(FLASHOCC_OPENCV_INCLUDE opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
find_library(FLASHOCC_OPENCV_CORE opencv_core REQUIRED)
find_library(FLASHOCC_OPENCV_IMGPROC opencv_imgproc REQUIRED)
find_library(FLASHOCC_OPENCV_IMGCODECS opencv_imgcodecs REQUIRED)
add_executable(flashocc_insight insight.cpp)
target_include_directories(flashocc_insight SYSTEM PRIVATE ${FLASHOCC_OPENCV_INCLUDE})
target_link_libraries(flashocc_insight PRIVATE flashocc_pool SimaNeat::sima_neat
  ${FLASHOCC_OPENCV_CORE} ${FLASHOCC_OPENCV_IMGPROC} ${FLASHOCC_OPENCV_IMGCODECS})
target_compile_options(flashocc_insight PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_insight test_insight.cpp)
target_include_directories(test_insight SYSTEM PRIVATE ${FLASHOCC_OPENCV_INCLUDE})
target_link_libraries(test_insight PRIVATE flashocc_pool SimaNeat::sima_neat
  ${FLASHOCC_OPENCV_CORE} ${FLASHOCC_OPENCV_IMGPROC} ${FLASHOCC_OPENCV_IMGCODECS})
add_executable(check_preprocess check_preprocess.cpp)
target_include_directories(check_preprocess SYSTEM PRIVATE ${FLASHOCC_OPENCV_INCLUDE})
target_link_libraries(check_preprocess PRIVATE flashocc_pool SimaNeat::sima_neat
  ${FLASHOCC_OPENCV_CORE} ${FLASHOCC_OPENCV_IMGPROC} ${FLASHOCC_OPENCV_IMGCODECS})
add_executable(check_packed_preprocess ../experiments/packing/check_packed_preprocess.cpp)
target_include_directories(check_packed_preprocess PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(check_packed_preprocess SYSTEM PRIVATE ${FLASHOCC_OPENCV_INCLUDE})
target_link_libraries(check_packed_preprocess PRIVATE flashocc_pool SimaNeat::sima_neat
  ${FLASHOCC_OPENCV_CORE} ${FLASHOCC_OPENCV_IMGPROC} ${FLASHOCC_OPENCV_IMGCODECS})
add_executable(check_camera_batch ../experiments/batch6/check_camera_batch.cpp)
target_include_directories(check_camera_batch PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(check_camera_batch PRIVATE flashocc_pool SimaNeat::sima_neat)
target_compile_options(check_camera_batch PRIVATE -Wall -Wextra -Wpedantic)
add_executable(check_camera_transport ../experiments/batch6/check_camera_transport.cpp)
target_include_directories(check_camera_transport PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(check_camera_transport PRIVATE flashocc_pool SimaNeat::sima_neat)
target_compile_options(check_camera_transport PRIVATE -Wall -Wextra -Wpedantic)
add_executable(profile_packed_camera ../experiments/packing/profile_packed_camera.cpp)
target_include_directories(profile_packed_camera PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(profile_packed_camera PRIVATE flashocc_pool SimaNeat::sima_neat)
target_compile_options(profile_packed_camera PRIVATE -Wall -Wextra -Wpedantic)
