cmake_minimum_required(VERSION 3.16)
project(flashocc_pool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(flashocc_pool bev_pool.cpp occupancy_labels.cpp occupancy_raycast.cpp)
target_include_directories(flashocc_pool PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(flashocc_pool PUBLIC Threads::Threads)
target_compile_options(flashocc_pool PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(pool_bench pool_bench.cpp)
target_link_libraries(pool_bench PRIVATE flashocc_pool)
target_compile_options(pool_bench PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME pool_contracts COMMAND pool_bench --self-test)
add_executable(test_labels test_labels.cpp)
target_link_libraries(test_labels PRIVATE flashocc_pool)
add_test(NAME occupancy_labels COMMAND test_labels)
add_executable(test_raycast test_raycast.cpp)
target_link_libraries(test_raycast PRIVATE flashocc_pool)
add_test(NAME occupancy_raycast COMMAND test_raycast)
add_executable(raycast_volume raycast_volume.cpp)
target_link_libraries(raycast_volume PRIVATE flashocc_pool)
